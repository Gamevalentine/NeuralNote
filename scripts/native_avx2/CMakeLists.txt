cmake_minimum_required(VERSION 3.24)
project(notiq_isa_quickcheck LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(notiq_isa_baseline cpu_isa_smoke.cpp)
target_include_directories(notiq_isa_baseline PRIVATE ../../notiq-native)
add_test(NAME notiq_isa_baseline COMMAND notiq_isa_baseline baseline)
add_executable(notiq_isa_avx2 cpu_isa_smoke.cpp)
target_include_directories(notiq_isa_avx2 PRIVATE ../../notiq-native)
target_compile_definitions(notiq_isa_avx2 PRIVATE NOTIQ_EXPERIMENTAL_AVX2=1)
if(MSVC)
  target_compile_options(notiq_isa_avx2 PRIVATE /arch:AVX2)
endif()
add_test(NAME notiq_isa_diagnostic COMMAND notiq_isa_avx2 avx2)
