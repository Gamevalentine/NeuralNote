cmake_minimum_required(VERSION 3.24)
project(notiq_stage3_native_reference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(MUSCRIPTOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(MUSCRIPTOR_BUILD_BENCH OFF CACHE BOOL "" FORCE)
set(MUSCRIPTOR_VULKAN OFF CACHE BOOL "" FORCE)
set(MUSCRIPTOR_METAL OFF CACHE BOOL "" FORCE)
set(MUSCRIPTOR_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/muscriptor/cpp" muscriptor-native)
add_executable(notiq-native-reference "${CMAKE_CURRENT_SOURCE_DIR}/../../../notiq-native/main.cpp")
target_link_libraries(notiq-native-reference PRIVATE muscriptor_ggml)
